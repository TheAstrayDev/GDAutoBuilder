cmake_minimum_required(VERSION 3.21)
project(RhythmBuilder VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(${PROJECT_NAME} SHARED src/main.cpp src/GeneratorCore.cpp)
if(NOT DEFINED ENV{GEODE_SDK})
    message(FATAL_ERROR "Set GEODE_SDK to the Geode SDK directory")
endif()
if(WIN32 AND GEODE_COMP_GD_VERSION EQUAL 22091)
    include(compat/sdk-22091.cmake)
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
setup_geode_mod(${PROJECT_NAME})
